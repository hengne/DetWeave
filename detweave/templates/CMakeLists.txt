cmake_minimum_required(VERSION 3.16)
project(DetWeaveGeometry LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Geant4 REQUIRED gdml)
include(${Geant4_USE_FILE})
configure_file(materials.gdml materials.gdml COPYONLY)
add_executable(detweave-check check.cc geometry.cc)
target_link_libraries(detweave-check ${Geant4_LIBRARIES})
option(DETWEAVE_BUILD_SIMULATION "Build geometry-only geantino smoke simulation" ON)
if(DETWEAVE_BUILD_SIMULATION)
  add_executable(detweave-sim simulation.cc geometry.cc)
  target_link_libraries(detweave-sim ${Geant4_LIBRARIES})
endif()
